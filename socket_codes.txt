MAIN:
    MOV AL, 10000001b   ; AL me binary value 81h save ki
    ROR AL, 1           ; AL ko right side rotate kiya 1 bit (Ab AL = C0h ho gaya)
    
    MOV BL, AL          ; Final AL ki value ko BL me safe rakh liya baad ke liye

    ; --- UPPER part ('C') ko isolate aur print karne ke liye ---
    MOV AL, BL          ; BL se value wapas AL me laye
    SHR AL, 4           ; Right shift kiya 4 bits taake shuru ka 'C' bache (AL = 0Ch)
    CALL PRINT_HEX_DIGIT

    ; --- LOWER part ('0') ko isolate aur print karne ke liye ---
    MOV AL, BL          ; BL se value phir se AL me laye
    AND AL, 0Fh         ; Masking ki taake sirf aakhri ka '0' bache (AL = 00h)
    CALL PRINT_HEX_DIGIT

    ; --- Program Khatam ---
    MOV AH, 4Ch         ; DOS exit function
    INT 21h

; --- Subroutine: AL ke 4-bit number ko ASCII character me badal kar print karega ---
PRINT_HEX_DIGIT:
    CMP AL, 9           ; Check karo ke number 9 se chota hai ya bada
    JBE ADD_ZERO        ; Agar 9 ya usse chota hai, to seedha ADD_ZERO par jump karo
    ADD AL, 7           ; Agar A-F ke beech hai, to 7 add karo taake ASCII table me 'A' tak pohanch sake
ADD_ZERO:
    ADD AL, '0'         ; '0' (30h) add kiya taake proper ASCII value ban jaye
    
    ; DOS use karke character screen par print karna
    MOV DL, AL          ; Character ko DL register me dala kyunki printing ke liye ye zaroori hai
    MOV AH, 02h         ; DOS function 2: Screen par single character dikhane ke liye
    INT 21h             ; Interrupt call kiya
    RET                 ; Wapas main code me jao

org 100h          ; DOS COM file ka standard start

MAIN:
    ; --- ADD A and B ---
    MOV AL, A   ; Variable A ki value ko AL register me dala
    ADD AL, B   ; AL me variable B ki value ko plus (add) kiya
    MOV RES_ADD, AL ; Result ko RES_ADD variable me save kar diya

    ; --- SUB B from A ---
    MOV AL, A   ; AL register me dobara A ki value ko dala
    SUB AL, B   ; AL mese B ki value ko minus (subtract) kiya (A - B)
    MOV RES_SUB, AL ; Result ko RES_SUB variable me save kar diya


    ; =========================================================
    ; PRINTING SECTION (Screen par dikhane ke liye)
    ; =========================================================

    ; --- 1. ADD ka result print karein ---
    MOV AL, RES_ADD   ; ADD ka result AL me dala
    CALL PRINT_HEX_BYTE ; Isko print karne ke liye subroutine call ki
    CALL PRINT_SPACE    ; Beech me thodi jagah (space) dene ke liye

    ; --- 2. SUB ka result print karein ---
    MOV AL, RES_SUB   ; SUB ka result AL me dala
    CALL PRINT_HEX_BYTE ; Isko print karne ke liye subroutine call ki

    HLT                 ; Program ko yahan rok diya (Halt)


; =========================================================
; SUBROUTINES (Funtions)
; =========================================================

; --- Pure 8-bit Byte ko Hex me print karne ka function ---
PRINT_HEX_BYTE:
    MOV BL, AL          ; AL ki asli value BL me safe kar li
    
    ; Shuru ka 4-bit (Upper Nibble) print karo
    SHR AL, 4           ; Right shift 4 bits (AL = 01h ho jayega agar 1Ah ho)
    CALL PRINT_DIGIT    ; Ek digit print karne wale function ko call kiya
    
    ; Aakhri ka 4-bit (Lower Nibble) print karo
    MOV AL, BL          ; Value wapas AL me laye
    AND AL, 0Fh         ; Sirf aakhri 4 bits bachaye (0Fh filter use kiya!)
    CALL PRINT_DIGIT    ; Doosra digit print kiya
    RET

; --- Single 4-bit Digit ko ASCII me badal kar print karne ka function ---
PRINT_DIGIT:                                        
    CMP AL, 9           ; Check karo 9 se bada hai ya chota
    JBE ADD_ZERO        ; Agar 9 ya chota hai to jump karo
    ADD AL, 7           ; Agar A-F hai to 7 add karo gap khatam karne ke liye
ADD_ZERO:
    ADD AL, '0'         ; '0' (30h) add kiya ASCII banane ke liye
    MOV DL, AL          ; Character DL me dala
    MOV AH, 02h         ; DOS print character function
    INT 21h             ; Screen par dikhao
    RET

; --- Do answers ke beech me space print karne ka function ---
PRINT_SPACE:
    MOV DL, ' '         ; Space character DL me dala
    MOV AH, 02h         ; DOS print function
    INT 21h
    RET

Print_Decimal_Dynamic macro hexVal
    ; Registers ka data safe karne ke liye variables use kiye (No Stack)
    MOV temp_ax, AX
    MOV temp_bx, BX
    MOV temp_dx, DX

    MOV AL, hexVal       ; Number ko AL me load kiya
    MOV AH, 0            ; AH clear kiya (AX = 0000 8-bit number)
    
    MOV BL, 10           ; Divisor = 10

    ; --- Step 1: Pehla Division (Aakhri digit ke liye) ---
    DIV BL               ; AX / 10 -> AL = Quotient, AH = Remainder
    MOV d3, AH           ; Aakhri digit ko d3 me save kiya (Units)
    
    ; --- Step 2: Doosra Division (Beech ke digit ke liye) ---
    MOV AH, 0            ; Clear AH
    DIV BL               ; AX / 10 -> AL = Hundred's, AH = Ten's
    MOV d2, AH           ; Beech ke digit ko d2 me save kiya (Tens)
    
    ; --- Step 3: bacha hua Quotient hi pehla digit hai ---
    MOV d1, AL           ; Pehle digit ko d1 me save kiya (Hundreds)

    ; --- Step 4: Simple Line-by-Line Printing (No Loop) ---
    ; 1st Digit Print Karo
    MOV DL, d1
    ADD DL, '0'          ; ASCII me badla

    INT 21h

    MOV AH, 02h

    ; 3rd Digit Print Karo

    MOV AH, 02h
    INT 21h

    ; Ek space print karo dono numbers ke beech me

    MOV AH, 02h
    INT 21h

    ; Registers ka data bina stack ke wapas le aaye
    MOV AX, temp_ax
    MOV BX, temp_bx
    MOV DX, temp_dx

; Ye macro AX register ke 16-bit number ko 5 decimal digits me torta hai.
CONVERT_TO_DEC16 MACRO
    ; --- 1. Ten Thousands Digit (10000) ---
    MOV DX, 0           ; DX:AX doubleword division ke liye DX ko clear kiya
    MOV BX, 10000       ; BX me 10000 dala
    DIV BX              ; DX:AX / BX -> AX = Quotient, DX = Remainder
    MOV CX, DX          ; Remainder ko CX me safe kar liya
    MOV DL, AL          ; Ten Thousands digit ko DL me dala
    ADD DL, '0'         ; ASCII me badla
    MOV AH, 02h         ; DOS print function
    INT 21h

    ; --- 2. Thousands Digit (1000) ---
    MOV AX, CX          ; Bacha hua remainder AX me laya
    MOV DX, 0           ; DX clear kiya
    MOV BX, 1000        ; BX me 1000 dala
    DIV BX              ; AX = Quotient, DX = Remainder
    MOV CX, DX          ; Remainder ko CX me safe kiya
    MOV DL, AL          ; Thousands digit DL me dala
    ADD DL, '0'         
    MOV AH, 02h         
    INT 21h

    ; --- 3. Hundreds Digit (100) ---
    MOV AX, CX          ; Remainder AX me laya
    MOV DX, 0
    MOV BX, 100         ; BX me 100 dala
    DIV BX              ; AX = Quotient, DX = Remainder
    MOV CX, DX          ; Remainder ko CX me safe kiya
    MOV DL, AL          ; Hundreds digit DL me dala
    ADD DL, '0'         
    MOV AH, 02h         
    INT 21h

    ; --- 4. Tens aur Ones Digits (10) ---
    MOV AX, CX          ; Remainder AX me laya
    MOV BL, 10          ; Ab sirf 8-bit division kaafi hai (AX / BL)

    MOV CL, AH          ; Ones digit ko CL me safe kiya

    ADD DL, '0'         
    MOV AH, 02h         
    INT 21h

    ; Ones print karo
    MOV DL, CL          
    ADD DL, '0'         
    MOV AH, 02h         
    INT 21h
ENDM

///////addition using stack and procedure/////////////
.MODEL SMALL
.STACK 100H
.DATA
    msg    DB 'The Sum is: $'
    result DW ?                  ; Result ko store karne ke liye variable
    
    ; Result ko decimal character mein badalne ke liye temporary variables
    d1     DB ?
    d2     DB ?

.CODE
MAIN PROC
    MOV AX, @DATA
    MOV DS, AX

    ; --- Step 1: Push two numbers onto the stack ---
    MOV AX, 5
    PUSH AX                      ; First Number (5) pushed
    
    MOV AX, 8
    PUSH AX                      ; Second Number (8) pushed

    ; --- Step 2: Call the procedure ---
    CALL add_numbers

    ; --- Step 3: After return, pop result into a variable ---
    POP result                   ; Procedure ne jo result push kiya tha, use variable me nikala

    ; --- Step 4: Display the result ---
    ; Display Text Message
    LEA DX, msg
    MOV AH, 09H
    INT 21H

    ; Numbers ko decimal mein convert karke print karne ka simple tareeka (5+8 = 13)
    MOV AX, result
    MOV BL, 10
    DIV BL                       ; AX / 10 -> AL = 1 (Tens), AH = 3 (Units)
    
    MOV d1, AL                   ; Tens digit '1'
    MOV d2, AH                   ; Units digit '3'

    ; Print Tens Digit
    MOV DL, d1
    ADD DL, '0'
    MOV AH, 02H
    INT 21H

    ; Print Units Digit
    MOV DL, d2
    ADD DL, '0'
    MOV AH, 02H
    INT 21H

exit_program:
    ; DOS Terminate
    MOV AH, 4CH
    INT 21H
MAIN ENDP


; =======================================================
; PROCEDURE: ADD TWO NUMBERS USING STACK MANAGEMENT
; =======================================================
add_numbers PROC
    ; Stack Order right now (Top to Bottom):
    ; [ Return Address (IP) ]  <- Current SP points here
    ; [ Second Number (8)   ]
    ; [ First Number (5)    ]

    POP BX                       ; 1. Pehle Return Address ko BX register mein nikal kar safe kiya
    
    POP CX                       ; 2. Second Number (8) ko CX mein nikala
    POP AX                       ; 3. First Number (5) ko AX mein nikala

    ADD AX, CX                   ; 4. Dono ko add kiya (5 + 8 = 13). Result AX mein aaya.

    PUSH AX                      ; 5. Result (13) ko stack par push kiya

    PUSH BX                      ; 6. Return Address (BX) ko wapas TOP par push kiya taaki RET kaam kar sake
    
    RET                          ; 7. Return to MAIN
add_numbers ENDP

END MAIN

    result  dw 0                      ; Final maximum number ko save karne ke liye 16-bit variable

; CODE SECTION: ACTUAL PROGRAM LOGIC
; =======================================================
.CODE
main PROC
    ; --- Data Segment (DS) ko Initialize karna ---
    mov ax, @data                     ; Data segment ka base address AX mein load kiya
    mov ds, ax                        ; Use DS register mein transfer kiya (Zaroori step)

    ; --- Loop aur Indexing ki Shuruat ---
    mov cx, n                         ; Counter register CX = 6 set kiya (Pore array ke liye)
    mov si, 0                         ; Source Index SI = 0 kiya (Array ke pehle element par point karne ke liye)

    ; --- Pehle Element ko Maximum Maanna ---
    mov al, [array+si]                ; Pehle element (5) ko uthakar AL mein dala aur Maan liya yahi Max hai
    inc si                            ; SI ko 1 se badhaya (Ab SI point kar raha hai doosre element yani 9 par)

find_max:
    ; --- Array ke baaki elements se Compare karne ka loop ---
    mov bl, [array+si]                ; Agla element (SI jahan point kar raha hai) use BL register mein load kiya
    cmp bl, al                        ; BL (New Element) ko AL (Current Max) se compare kiya (BL - AL)
    jle skip_update                   ; Jump if Less or Equal: Agar naya element chota ya barabar hai, toh update mat karo

    ; --- Agar naya element bada hai, toh Max update karo ---
    mov al, bl                        ; Agar BL bada hai, toh use AL mein daal do (Ab naya Max AL mein aa gaya)

skip_update:
    inc si                            ; Index (SI) ko 1 se badhao taaki agle element par ja sakein
    loop find_max                     ; Loop automatic CX ko 1 kam karega. Agar CX != 0, toh 'find_max' par wapas koodega

    ; =======================================================
    ; LOOP KHATAM: Ab AL register ke andar sabse bada number (13) hai
    ; =======================================================

    mov ah, 0                         ; AH register ko 0 kiya taaki AX mein koi purana kachra (garbage) na rahe
    mov result, ax                    ; Final maximum value (AX) ko 'result' variable mein save kar diya

    ; --- Result ko Decimal mein Print karne ke liye Function Call ---
    call display_result               ; 'display_result' procedure ko pukara (call kiya)

    ; --- DOS Terminal se Program safely exit karne ka code ---
    mov ah, 4Ch                       ; DOS terminate program function
    int 21h                           ; Control Operating System ko wapas de diya
main ENDP


; =======================================================
; PROCEDURE: 16-BIT REGISTER VALUE KO DECIMAL MEIN PRINT KARNA
; =======================================================
display_result PROC
    ; --- Newline print karo ---
    lea dx, newline                   ; Newline string ka address DX mein load kiya
    mov ah, 09h                       ; DOS String display function
    int 21h                           ; Agli khali line par jaane ke liye execute kiya

    ; --- Binary/Hex se Decimal Conversion Logic (Using Division) ---
    mov ax, result                    ; 'result' variable (13) ko wapas AX register mein load kiya
    mov cx, 0                         ; CX = 0 kiya (Yeh ginega ki total kitne digits bane hain)
    lea si, buffer+5                  ; Buffer ke aakhri kone (End) par SI ko set kiya kyunki digits ulte store hote hain

convert_loop: 
    xor dx, dx                        ; DX ko completely 0 kiya (Division se pehle DX:AX clear hona zaroori hai)
    mov bx, 10                        ; Divisor BL/BX = 10 set kiya
    div bx                            ; DX:AX ko 10 se divide kiya -> AL = Quotient (1), DX = Remainder (3)
    
    add dl, '0'                       ; Remainder (DL) mein '0' (48) add karke use ASCII character banaya ('3')
    dec si                            ; Buffer mein ek kadam peeche (Left side) move kiya
    mov [si], dl                      ; Uskis jagah par ASCII character save kar diya
    inc cx                            ; Digit counter (CX) ko 1 se badha diya
    
    cmp ax, 0                         ; Check kiya kya abhi bhi Quotient (AX) bacha hai?
    jne convert_loop                  ; Jump if Not Equal: Agar AX abhi 0 nahi hua, toh dobara divide karo

    ; --- Convert kiye huye Digits ko Screen par Print karna ---
    mov ah, 02h                       ; DOS Single character display function set kiya
print_loop: 
    mov dl, [si]                      ; Buffer se ek character (Pehle Tens fir Units) DL mein uthaya
    int 21h                           ; Use screen par print kar diya
    inc si                            ; SI ko aage badhaya agle digit ke liye
    loop print_loop                   ; Jitne digits bane the (CX), utni baar chala kar loop khatam kiya

    ret                               ; Procedure se wapas main program mein jane ke liye
display_result ENDP

end main













//////5 numbers array all in one user input max min sum///////
ORG 100h

.DATA
    ; User se inputs lene ke liye messages
    msg_input DB 'Enter 5 single-digit numbers (0-9): ', 0Dh, 0Ah, '$'
    msg_max   DB 0Dh, 0Ah, 'Maximum Number is: $'
    msg_min   DB 0Dh, 0Ah, 'Minimum Number is: $'
    msg_sum   DB 0Dh, 0Ah, 'Sum of Elements is: $'
    
    ; 5 elements ka khali array aur size definition
    array     DB 5 DUP(0)
    n         EQU 5

.CODE
main PROC
    MOV AX, @DATA
    MOV DS, AX

    ; --- 1. User ko Input Prompt dikhao ---
    LEA DX, msg_input
    MOV AH, 09h
    INT 21h

    ; --- 2. Array Input Loop (User se 5 numbers lena) ---
    MOV CX, n
    MOV SI, 0
input_loop:
    MOV AH, 01h              ; Single character input (Result AL mein aayega)
    INT 21h
    
    SUB AL, '0'              ; ASCII character ko actual number mein badla
    MOV [array+SI], AL       ; Number ko array mein save kiya
    INC SI
    LOOP input_loop

    ; --- 3. Initial Setup: Pehle Element ko hi Max aur Min Maan lo ---
    MOV SI, 0
    MOV AL, [array+SI]       ; AL ko Max bana diya
    MOV AH, [array+SI]       ; AH ko Min bana diya
    MOV BL, [array+SI]       ; BL mein Sum (Total) start kiya
    
    INC SI
    MOV CX, n
    DEC CX                   ; Pehla element check ho gaya, ab bache 4 elements

; --- 4. Main Processing Loop (Find Max, Min, and Sum) ---
process_loop:
    MOV DL, [array+SI]       ; Current element DL mein load kiya
    
    ; Sum Calculation
    ADD BL, DL               ; BL = BL + DL (Total Sum accumulator)

    ; Max Check
    CMP DL, AL
    JLE check_min            ; Agar DL <= AL, toh Max update skip karo
    MOV AL, DL               ; Naya Max AL mein save kiya
    JMP skip_min             ; Agar naya max mil gaya, toh woh min nahi ho sakta

check_min:
    ; Min Check
    CMP DL, AH
    JGE skip_min             ; Agar DL >= AH, toh Min update skip karo
    MOV AH, DL               ; Naya Min AH mein save kiya

skip_min:
    INC SI
    LOOP process_loop

    ; --- Registers state right now: AL = Max, AH = Min, BL = Sum ---
    ; Inhe registers mein safe rakhne ke liye move karte hain
    MOV CH, AL               ; CH = Max
    MOV CL, AH               ; CL = Min
    MOV BH, BL               ; BH = Sum

    ; =======================================================
    ; DISPLAY SECTION (Using Bit Manipulation / Direct Print)
    ; =======================================================

    ; --- 1. MAX PRINT KARNA ---
    LEA DX, msg_max
    MOV AH, 09h
    INT 21h
    MOV DL, CH               ; Max value DL mein layi
    ADD DL, '0'              ; ASCII banaya
    MOV AH, 02h
    INT 21h

    ; --- 2. MIN PRINT KARNA ---
    LEA DX, msg_min
    MOV AH, 09h
    INT 21h
    MOV DL, CL               ; Min value DL mein layi
    ADD DL, '0'              ; ASCII banaya
    MOV AH, 02h
    INT 21h

    ; --- 3. SUM PRINT KARNA (Can be 2 digits, e.g., 9+9+9+9+9 = 45) ---
    LEA DX, msg_sum
    MOV AH, 09h
    INT 21h

    MOV AL, BH               ; Sum value AX mein layi (Kyunki BH mein total sum tha)
    MOV AH, 0
    MOV DL, 10
    DIV DL                   ; AX / 10 -> AL = Tens Digit, AH = Units Digit

    MOV BX, AX               ; Result register mein save kiya (BL = Tens, BH = Units)

    ; Tens Digit Print (Using dynamic check)
    CMP BL, 0
    JE print_units_sum       ; Agar Tens digit 0 hai (Sum < 10), toh direct units par jao
    MOV DL, BL
    ADD DL, '0'
    MOV AH, 02h
    INT 21h

print_units_sum:
    ; Units Digit Print
    MOV DL, BH
    ADD DL, '0'
    MOV AH, 02h
    INT 21h

exit_program:
    ; --- Exit Program ---
    MOV AH, 4Ch
    INT 21h
main ENDP
END main








/////average array///////////
                                                                        ORG 100h

.DATA
    array   DB 4, 8, 3, 9, 6          ; 5 numbers ka array
    n       DB 5                      ; Array ka size (N = 5)
    msg_avg DB 'Average is: $'

.CODE
main PROC
    MOV AX, @DATA
    MOV DS, AX

    MOV CL, n                         ; Counter CX mein dala
    MOV CH, 0
    MOV SI, 0
    MOV BL, 0                         ; BL ko Sum accumulator banaya (Sum = 0)

sum_loop:
    ADD BL, [array+SI]                ; Har element ko BL mein plus karte gye
    INC SI
    LOOP sum_loop

    ; --- Average = Total Sum / N ---
    MOV AL, BL                        ; Total Sum ko AL mein dala
    MOV AH, 0                         ; AH clear kiya division ke liye
    DIV n                             ; AX / n -> AL = Average (Quotient)

    MOV BL, AL                        ; Average ko BL mein safe kar liya

    ; --- Print Message ---
    LEA DX, msg_avg
    MOV AH, 09h
    INT 21h

    ; --- Direct Print (Single Digit Average) ---
    MOV DL, BL
    ADD DL, '0'                       ; ASCII mein badla
    MOV AH, 02h
    INT 21h                           ; Screen par print kiya (e.g., 30 / 5 = 6)

exit1:
    MOV AH, 4Ch
    INT 21h
main ENDP
END main

//////search in array////////
ORG 100h

.DATA
    array     DB 5, 3, 8, 2, 9        ; Array jisme search karna hai
    n         EQU 5
    search_for DB 2                   ; Jis number ko dhoondna hai (Hum '2' dhoond rahe hain)

    msg_not   DB 'Number Not Found!$'

.CODE
main PROC
    MOV AX, @DATA
    MOV DS, AX

    MOV CX, n                         ; Counter = 5
    MOV SI, 0
    MOV AL, search_for                ; AL mein woh number dala jo dhoondna hai

search_loop:
    CMP [array+SI], AL                ; Array ke current element ko AL se compare kiya
    JE number_found                   ; JUMP IF EQUAL: Agar match ho gaya, toh loop se bahar kood jao
    
    INC SI                            ; Agar nahi mila, toh agle element par jao

    ; --- Agar loop poora chal gaya aur match nahi hua, toh yeh chalega ---
    LEA DX, msg_not
    MOV AH, 09h
    INT 21h
    JMP exit3                         ; Direct exit par jao taaki found wala msg skip ho jaye

number_found:
    ; --- Agar JE call hua toh control yahan aayega ---
    LEA DX, msg_found
    MOV AH, 09h
    INT 21h

exit3:
    MOV AH, 4Ch
    INT 21h
main ENDP
END main
